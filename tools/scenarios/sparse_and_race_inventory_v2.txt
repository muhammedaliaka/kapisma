player Bot1 spawn
player Bot2 spawn
wait 2
kapisma game BINGO
kapisma origins off
kapisma rule BINGO borderStart 200
kapisma rule BINGO tasks 0
kapisma rule BINGO verifySec 30
kapisma start
await Bu seed ve sınırda 25 farklı hedef doğrulanamadı. Maç başlatılmadı.
await BINGO kapandi
kapisma game ESYA
kapisma origins off
kapisma rule ESYA borderStart 200
kapisma rule ESYA roundSec 90
kapisma rule ESYA roundLimit 3
kapisma rule ESYA keepInventory 1
kapisma start
await Eşya Yarışı · Tur 1 başladı
kapisma debug race Bot1
capture first race Bot1 round=1 target=(minecraft:(?:dirt|sand|gravel)) known=true baseline=0 count=0 score=0 lives=5
kapisma debug race Bot2
capture second race Bot2 round=1 target=((?!%{first} )minecraft:(?:dirt|sand|gravel)) known=true baseline=0 count=0 score=0 lives=5
give Bot1 %{first} 1
wait 1
kapisma debug race Bot1
capture initial_score race Bot1 round=1 target=%{first} known=true baseline=0 count=1 score=(1) lives=5 skips=0 resolved=true
give Bot1 minecraft:dirt 1
give Bot1 minecraft:sand 1
give Bot1 minecraft:gravel 1
player Bot1 kill
await Bot1 left the game
give Bot2 %{second} 1
await Eşya Yarışı · Tur 2 başladı
kapisma debug race Bot1
capture next race Bot1 round=2 target=(minecraft:(?:dirt|sand|gravel)) known=false baseline=0 count=-1 score=1 lives=5 skips=0 resolved=false
player Bot1 spawn
await Bot1 joined the game
wait 2
kapisma debug race Bot1
capture positive race Bot1 round=2 target=%{next} known=true baseline=([1-9]\d*) count=\1 score=1 lives=5 skips=0 resolved=false
damage Bot1 1000 minecraft:out_of_world
await Bot1 left the game
player Bot1 spawn
await Bot1 joined the game
wait 2
kapisma debug race Bot1
capture after_death race Bot1 round=2 target=%{next} known=true baseline=%{positive} count=%{positive} score=(1) lives=5 skips=0 resolved=false
give Bot1 %{next} 1
wait 1
kapisma debug race Bot1
capture gained race Bot1 round=2 target=%{next} known=true baseline=%{positive} count=\d+ score=(2) lives=5 skips=0 resolved=true
kapisma stop
