# Güç ve direnç yalnızca gizli role açılır, alana etki veya parçacık gönderilmez.
player Bot1 spawn
player Bot2 spawn
player Bot3 spawn
gamerule naturalRegeneration false
kapisma game KATIL
kapisma origins off
kapisma rule KATIL graceSec 0
kapisma start
await Katil Kim başladı!
gamerule doMobSpawning false
execute as @a run katilgucu
execute as @a run damage @s 5 minecraft:generic
execute if entity @a[nbt={Health:16.0f}] run say HIDDEN_RESISTANCE_OK
expect HIDDEN_RESISTANCE_OK
execute as Bot1 at @s run summon minecraft:pig ~ ~2 ~ {Tags:["power_probe_1"],NoAI:1b,NoGravity:1b}
execute as Bot2 at @s run summon minecraft:pig ~ ~2 ~ {Tags:["power_probe_2"],NoAI:1b,NoGravity:1b}
execute as Bot3 at @s run summon minecraft:pig ~ ~2 ~ {Tags:["power_probe_3"],NoAI:1b,NoGravity:1b}
damage @e[tag=power_probe_1,limit=1] 3 minecraft:player_attack by Bot1
damage @e[tag=power_probe_2,limit=1] 3 minecraft:player_attack by Bot2
damage @e[tag=power_probe_3,limit=1] 3 minecraft:player_attack by Bot3
execute if entity @e[type=minecraft:pig,nbt={Health:4.0f},tag=power_probe_1] run say HIDDEN_STRENGTH_OK
execute if entity @e[type=minecraft:pig,nbt={Health:4.0f},tag=power_probe_2] run say HIDDEN_STRENGTH_OK
execute if entity @e[type=minecraft:pig,nbt={Health:4.0f},tag=power_probe_3] run say HIDDEN_STRENGTH_OK
expect HIDDEN_STRENGTH_OK
execute as @a run katilgucu
gamerule naturalRegeneration true
kapisma stop
